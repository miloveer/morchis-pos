    match /inventario/operacion/insumos/{insumoId} {
      function fichaValida(i) {
        let campos = ['nombre', 'tienda', 'presentacion', 'stock', 'minimo', 'comprado',
          'registradoPor', 'registradoNombre', 'registradoColor', 'registradoEn'];
        return i.keys().hasOnly(campos) && i.keys().hasAll(campos)
          && i.nombre is string && i.nombre.size() > 0 && i.nombre.size() <= 120
          && i.tienda is string && i.tienda.size() > 0 && i.tienda.size() <= 120
          && i.presentacion is string && i.presentacion.size() > 0 && i.presentacion.size() <= 80
          && i.stock is number && i.stock >= 0 && i.stock <= 1000000
          && i.minimo is number && i.minimo >= 0 && i.minimo <= 1000000
          && i.comprado is bool;
      }
      function firmaActual(i) {
        let datos = profile();
        let nombre = datos.get('nombre', '');
        let color = datos.get('colorInventario', 'gris');
        return i.registradoPor == request.auth.uid
          && i.registradoNombre == (nombre == '' ? 'Personal' : nombre)
          && i.registradoColor == (color in ['rojo', 'morado', 'rosa', 'amarillo', 'gris'] ? color : 'gris')
          && i.registradoEn == request.time;
      }
      allow read: if staff();
      allow create: if staff() && fichaValida(request.resource.data)
        && request.resource.data.comprado == false && firmaActual(request.resource.data);
      allow update: if staff() && fichaValida(request.resource.data) && (
        (request.resource.data.comprado == false && firmaActual(request.resource.data))
        || (resource.data.minimo > resource.data.stock
          && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['comprado']))
      );
    }
